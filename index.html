4) Replace your flashHit function with this stronger version

function flashHit(color, opacity = 1, duration = 240) {
  ui.hitFlash.style.background = color;
  ui.hitFlash.style.opacity = String(opacity);
  ui.hitFlash.style.transition = 'none';
  void ui.hitFlash.offsetWidth;

5) Add premium helpers under your utility functions

function kickHud(el) {
  if (!el) return;
  el.classList.remove('premium-pop');
  void el.offsetWidth;
  el.classList.add('premium-pop');
}

function addHitStop(ms) {
  hitStopUntil = Math.max(hitStopUntil, performance.now() + ms);
}

function addCameraPunch(amount = 0.5, zoomMs = 140) {
  cameraShake = Math.max(cameraShake, amount);
  boostFovUntil = Math.max(boostFovUntil, performance.now() + zoomMs);
}

function premiumTrailColor() {
  if (isOverdrive) return new THREE.Color().setHSL((performance.now() * 0.001) % 1, 1, 0.62);
  if (combo >= 30) return new THREE.Color('#ffd60a');
  if (combo >= 12) return new THREE.Color('#00f5d4');
  return new THREE.Color(currentTheme.gridColor);
}

function spawnTrail(now) {
  if (trailParticles.length >= BALANCE.trailMax) return;
  if (now - lastTrailSpawn < 1000 / (30 * BALANCE.trailSpawnRate)) return;
  lastTrailSpawn = now;

  const trail = new THREE.Mesh(
    trailGeo,
    new THREE.MeshBasicMaterial({
      color: premiumTrailColor(),
      transparent: true,
      opacity: 0.5 + Math.min(combo / 120, 0.16)
    })
  );
  trail.position.copy(playerModel.position);
  trail.scale.setScalar(0.8 + Math.min(speedMultiplier * 0.12, 0.4));
  trail.life = 1;
  scene.add(trail);
  trailParticles.push(trail);
}
